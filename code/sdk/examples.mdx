## Bulk create resources

When you need to create many resources at once, loop over your input array, call `resources.create` for each item, and collect the results. This example batches requests concurrently using `Promise.all` with a concurrency limit so you don't overwhelm the API.

<Note>
  The Google API enforces a default rate limit of 100 requests per second. Adjust the `BATCH_SIZE` constant to stay comfortably below that ceiling for your tier. See [rate limits](/code/advanced/rate-limits) for details.
</Note>

```javascript title="bulkCreate.js"
import { GoogleClient } from "@google/sdk";

const client = new GoogleClient({ apiKey: process.env.GOOGLE_API_KEY });

const BATCH_SIZE = 10; // number of concurrent requests

/**
 * Creates an array of resources in parallel batches.
 * @param {Array<{name: string, type: string, metadata: object}>} items
 * @returns {Promise<Array<{input: object, resource?: object, error?: Error}>>}
 */
async function bulkCreateResources(items) {
  const results = [];

  // Split the input into chunks of BATCH_SIZE
  for (let i = 0; i < items.length; i += BATCH_SIZE) {
    const batch = items.slice(i, i + BATCH_SIZE);

    const batchResults = await Promise.all(
      batch.map(async (item) => {
        try {
          const resource = await client.resources.create(item);
          return { input: item, resource };
        } catch (error) {
          console.error(`Failed to create "${item.name}":`, error.message);
          return { input: item, error };
        }
      })
    );

    results.push(...batchResults);
    console.log(`Processed batch ${Math.floor(i / BATCH_SIZE) + 1}: ${batchResults.length} items`);
  }

  const succeeded = results.filter((r) => r.resource).length;
  const failed = results.filter((r) => r.error).length;
  console.log(`Done. ${succeeded} created, ${failed} failed.`);

  return results;
}

// Example usage
const itemsToCreate = [
  { name: "Resource A", type: "document", metadata: { author: "alice@example.com" } },
  { name: "Resource B", type: "document", metadata: { author: "bob@example.com" } },
  { name: "Resource C", type: "snippet", metadata: { language: "python" } },
];

bulkCreateResources(itemsToCreate).then((results) => {
  const ids = results.filter((r) => r.resource).map((r) => r.resource.id);
  console.log("Created IDs:", ids);
});
```

## Paginate all resources

Use a `while` loop with a cursor to walk through every page of results until `has_more` is `false`. This pattern works regardless of how many total records exist — you never need to know the total count upfront.

```python title="paginate_all.py"
from google_sdk import GoogleClient
import os

client = GoogleClient(api_key=os.environ["GOOGLE_API_KEY"])


def fetch_all_resources(resource_type: str | None = None) -> list:
    """
    Fetches every resource from the API using cursor-based pagination.

    Args:
        resource_type: Optional type filter (e.g. "document", "snippet").

    Returns:
        A flat list of all resource objects.
    """
    all_resources = []
    cursor = None
    page_number = 0

    while True:
        page_number += 1

        # Build the list parameters; only include cursor if we have one
        params = {"limit": 100}
        if cursor:
            params["cursor"] = cursor
        if resource_type:
            params["type"] = resource_type

        page = client.resources.list(**params)
        all_resources.extend(page.data)

        print(f"Page {page_number}: fetched {len(page.data)} resources "
              f"(total so far: {len(all_resources)})")

        if not page.has_more:
            break

        cursor = page.next_cursor

    print(f"Finished. Total resources fetched: {len(all_resources)}")
    return all_resources


if __name__ == "__main__":
    resources = fetch_all_resources(resource_type="document")
    for r in resources:
        print(r.id, r.name)
```

## Handle webhook events

Google sends signed HTTP POST requests to your webhook URL when events occur. Verify the `X-Google-Signature` header before processing the payload to ensure the request is genuine. This example uses an Express.js handler.

<Warning>
  Always verify the webhook signature before acting on the event payload. Processing unverified webhooks can expose your application to spoofed events.
</Warning>

```javascript title="webhookHandler.js"
import express from "express";
import { verifyWebhookSignature, GoogleWebhookError } from "@google/sdk";

const app = express();

// Use the raw body parser for the webhook route so the signature check
// can compute an HMAC over the exact bytes that were sent.
app.post(
  "/webhooks/google",
  express.raw({ type: "application/json" }),
  async (req, res) => {
    const signature = req.headers["x-google-signature"];
    const secret = process.env.GOOGLE_WEBHOOK_SECRET;

    // 1. Verify the signature
    try {
      verifyWebhookSignature({
        payload: req.body,         // raw Buffer
        signature,
        secret,
      });
    } catch (err) {
      if (err instanceof GoogleWebhookError) {
        console.warn("Invalid webhook signature — request rejected.");
        return res.status(401).json({ error: "Invalid signature" });
      }
      throw err;
    }

    // 2. Parse the verified payload
    const event = JSON.parse(req.body.toString("utf8"));
    console.log("Received event:", event.type, "| ID:", event.id);

    // 3. Route on event type
    try {
      switch (event.type) {
        case "resource.created":
          await handleResourceCreated(event.data);
          break;

        case "resource.updated":
          await handleResourceUpdated(event.data);
          break;

        case "resource.deleted":
          await handleResourceDeleted(event.data);
          break;

        default:
          console.log("Unhandled event type:", event.type);
      }
    } catch (handlerError) {
      console.error("Handler error:", handlerError);
      // Return 500 so Google retries the delivery
      return res.status(500).json({ error: "Handler failed" });
    }

    // 4. Acknowledge receipt
    res.status(200).json({ received: true });
  }
);

async function handleResourceCreated(data) {
  console.log("New resource created:", data.id, data.name);
  // Add your business logic here
}

async function handleResourceUpdated(data) {
  console.log("Resource updated:", data.id);
  // Add your business logic here
}

async function handleResourceDeleted(data) {
  console.log("Resource deleted:", data.id);
  // Add your business logic here
}

app.listen(3000, () => console.log("Webhook server running on port 3000"));
```

## Retry on rate limit

When the API responds with `429 Too Many Requests`, read the `Retry-After` header to learn exactly how long to wait before trying again. This wrapper retries automatically up to a configurable maximum number of attempts.

```javascript title="retryOnRateLimit.js"
import { GoogleClient, GoogleRateLimitError } from "@google/sdk";

const client = new GoogleClient({ apiKey: process.env.GOOGLE_API_KEY });

const MAX_RETRIES = 4;

/**
 * Wraps an SDK call and automatically retries on 429 rate-limit responses.
 *
 * @template T
 * @param {() => Promise<T>} fn - A function that returns an SDK promise.
 * @param {number} [attempt=0] - Internal retry counter; do not pass this manually.
 * @returns {Promise<T>}
 */
async function withRateLimitRetry(fn, attempt = 0) {
  try {
    return await fn();
  } catch (err) {
    if (err instanceof GoogleRateLimitError && attempt < MAX_RETRIES) {
      // The Retry-After value is in seconds
      const retryAfterMs = (err.retryAfter ?? 1) * 1000;
      const jitter = Math.random() * 200; // add up to 200ms jitter

      console.warn(
        `Rate limited. Retrying in ${retryAfterMs + jitter}ms ` +
        `(attempt ${attempt + 1} of ${MAX_RETRIES})...`
      );

      await new Promise((resolve) => setTimeout(resolve, retryAfterMs + jitter));
      return withRateLimitRetry(fn, attempt + 1);
    }

    // Not a rate limit error, or we've exhausted retries — rethrow
    throw err;
  }
}

// Example: wrap any SDK call
async function main() {
  const resource = await withRateLimitRetry(() =>
    client.resources.get("res_abc123")
  );

  console.log("Fetched resource:", resource.id);

  // Works with any operation
  const created = await withRateLimitRetry(() =>
    client.resources.create({
      name: "Resilient resource",
      type: "document",
    })
  );

  console.log("Created:", created.id);
}

main().catch(console.error);
```

## Environment-aware client

Initialize different client instances for production and staging by reading an environment variable at startup. This pattern keeps configuration explicit and avoids accidentally running test workloads against the production API.

```javascript title="clientFactory.js"
import { GoogleClient } from "@google/sdk";

const ENVIRONMENTS = {
  production: {
    baseUrl: "https://api.google.com/v1",
    apiKey: process.env.GOOGLE_API_KEY_PROD,
  },
  staging: {
    baseUrl: "https://staging.api.google.com/v1",
    apiKey: process.env.GOOGLE_API_KEY_STAGING,
  },
  development: {
    baseUrl: "http://localhost:4000/v1",
    apiKey: process.env.GOOGLE_API_KEY_DEV ?? "dev-key-local",
  },
};

/**
 * Creates a Google SDK client configured for the current environment.
 * Reads GOOGLE_ENV (defaults to "development" if unset).
 *
 * @returns {GoogleClient}
 */
function createClient() {
  const env = process.env.GOOGLE_ENV ?? "development";
  const config = ENVIRONMENTS[env];

  if (!config) {
    throw new Error(
      `Unknown GOOGLE_ENV value: "${env}". ` +
      `Valid options are: ${Object.keys(ENVIRONMENTS).join(", ")}`
    );
  }

  if (!config.apiKey) {
    throw new Error(
      `API key is not set for environment "${env}". ` +
      `Set the corresponding environment variable before starting the app.`
    );
  }

  console.log(`Initializing Google SDK for environment: ${env}`);

  return new GoogleClient({
    apiKey: config.apiKey,
    baseUrl: config.baseUrl,
  });
}

// Export a single shared client for the current environment
export const client = createClient();
```

<Tip>
  Import `client` from this file throughout your application. When you deploy to a different environment, only the environment variables change — no code modifications required.
</Tip>
